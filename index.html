// Cloudflare Worker — budget-push + monthly cron // Secrets נדרשים: ONESIGNAL_APP_ID, ONESIGNAL_API_KEY, FIREBASE_URL, FIREBASE_SECRET export default { // HTTP requests (שליחת התראה מהאפליקציה) async fetch(request, env) { const cors = { 'Access-Control-Allow-Origin': '*', 'Access-Control-Allow-Methods': 'POST, OPTIONS', 'Access-Control-Allow-Headers': 'Content-Type', }; if (request.method === 'OPTIONS') return new Response(null, { headers: cors }); if (request.method !== 'POST') return new Response('Method not allowed', { status: 405, headers: cors }); let body; try { body = await request.json(); } catch { return new Response('Bad JSON', { status: 400, headers: cors }); } const { title = '💰 תקציב משפחתי', body: msg = 'הוצאה חדשה' } = body; const result = await sendOneSignal(title, msg, env); return new Response(JSON.stringify({ ok: true, result }), { headers: { ...cors, 'Content-Type': 'application/json' } }); }, // Cron trigger — רץ ב-10 לכל חודש בשעה 08:00 async scheduled(event, env, ctx) { ctx.waitUntil(runMonthlyJob(env)); } }; // ── חיוב חודשי אוטומטי ────────────────────────────────────── async function runMonthlyJob(env) { const fbBase = `${env.FIREBASE_URL}/budget`; const auth = `auth=${env.FIREBASE_SECRET}`; // קרא נתונים מ-Firebase const dataRes = await fetch(`${fbBase}.json?${auth}`); const data = await dataRes.json(); if (!data) return; const recurring = data.recurring ? Object.values(data.recurring) : []; const installments = data.installments ? Object.values(data.installments) : []; // חשב חודש נוכחי (תמיד ה-10 לחודש הוא תחילת חודש תקציבי) const now = new Date(); // אם היום ה-10 בדיוק, זה תחילת חודש חדש const y = now.getFullYear(); const m = now.getMonth() + 1; const monthKey = `${y}-${String(m).padStart(2, '0')}`; const updates = {}; // הוצאות קבועות const recApplied = data.months?.[monthKey]?.recurringApplied; if (!recApplied && recurring.length) { recurring.forEach(r => { const expId = `rec_${r.id}_${monthKey}`; const day = Math.min(r.day || 1, 28); const ts = new Date(y, m - 1, day, 8, 0, 0).getTime(); updates[`budget/months/${monthKey}/expenses/${expId}`] = { amount: r.amount, catId: r.catId, desc: r.name + ' (קבוע)', ts, recurring: true, payType: 'credit' }; }); updates[`budget/months/${monthKey}/recurringApplied`] = true; } // תשלומים const activeInst = installments.filter(i => i.active && i.paidCount < i.count); const finishedNames = []; for (const inst of activeInst) { const expId = `inst_${inst.id}_${monthKey}`; const existing = data.months?.[monthKey]?.expenses?.[expId]; if (existing) continue; const [sy, sm] = inst.startMonth.split('-').map(Number); const monthsDiff = (y - sy) * 12 + (m - sm); const paymentNum = monthsDiff + 1; if (paymentNum < 1 || paymentNum > inst.count || paymentNum <= inst.paidCount) continue; const isLast = paymentNum === inst.count; const ts = new Date(y, m - 1, Math.min(inst.day || 1, 28), 9, 0, 0).getTime(); updates[`budget/months/${monthKey}/expenses/${expId}`] = { amount: inst.monthly, catId: inst.catId, desc: `${inst.name} — תשלום ${paymentNum}/${inst.count}${isLast ? ' (אחרון!)' : ''}`, ts, payType: 'credit', installmentId: inst.id, paymentNum, isLast }; updates[`budget/installments/${inst.id}/paidCount`] = paymentNum; if (isLast) { updates[`budget/installments/${inst.id}/active`] = false; finishedNames.push(inst.name); } } // שמור ל-Firebase if (Object.keys(updates).length) { await fetch(`${env.FIREBASE_URL}/.json?${auth}`, { method: 'PATCH', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(updates) }); } // שלח התראה const recCount = recurring.length; const instCount = activeInst.length; let notifBody = `חודש חדש התחיל!`; if (recCount) notifBody += ` ${recCount} הוצאות קבועות נרשמו.`; if (instCount) notifBody += ` ${instCount} תשלומים עודכנו.`; if (finishedNames.length) notifBody += ` ✅ הסתיים: ${finishedNames.join(', ')}.`; await sendOneSignal('🗓️ חודש תקציבי חדש', notifBody, env); } async function sendOneSignal(title, body, env) { const res = await fetch('https://onesignal.com/api/v1/notifications', { method: 'POST', headers: { 'Content-Type': 'application/json', 'Authorization': `Key ${env.ONESIGNAL_API_KEY}`, }, body: JSON.stringify({ app_id: env.ONESIGNAL_APP_ID, included_segments: ['All'], headings: { he: title, en: title }, contents: { he: body, en: body }, ttl: 86400, }), }); return await res.json(); }
